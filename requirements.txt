# HEPTAPOD always-installed base requirements.
# Install with: pip install -r requirements.txt

# Bundle-specific deps live in toolkit.yaml under each bundle's `deps:`
# (pip-installed only when the bundle is selected, e.g.
# `tb install heptapod[inspire,pdg]`). See toolkit.yaml for the per-bundle
# breakdown.

# Orchestral framework — every heptapod tool subclasses BaseTool and uses
# RuntimeField / StateField; the toolbase serve host needs this to be
# import-able before any per-bundle tool module loads. 1.10 is the first
# release whose orchestral.mcp speaks MCP SDK 2.x, which is what that host
# talks to it over.
orchestral-ai>=1.10.0

# HTTP — tools/inspire/inspire_interface.py imports `requests` at module
# top level (and tools/inspire/__init__.py eagerly loads the interface).
# Kept in the always-installed base so the inspire bundle stays a freebie
# even on lean subset installs.
requests>=2.32.0

# YAML — the llp bundle parses kernel/geometry YAML unconditionally
# (tools/llp/llp_physics.py imports `yaml` at module top level), and
# bundle `deps:` only install when the bundle is named at install time,
# so envs that predate the bundle (or were installed without `[llp]`)
# would otherwise miss it. YAML data products are a base heptapod
# convention, not a bundle extra.
pyyaml>=6.0
